Creation & Manipulation
Array Creation and Manipulation Functions
অ্যারে তৈরি করা, উপাদান যোগ বা বাদ দেওয়ার ফাংশন।
array()
নতুন অ্যারে তৈরি করে।
$a = array(1,2,3);
array_push()
অ্যারের শেষে উপাদান যোগ করে।
array_push($a, 4, 5);
array_pop()
অ্যারের শেষ উপাদান সরিয়ে ফেরত দেয়।
$last = array_pop($a);
array_shift()
অ্যারের প্রথম উপাদান সরিয়ে ফেরত দেয়।
$first = array_shift($a);
array_unshift()
অ্যারের শুরুতে উপাদান যোগ করে।
array_unshift($a, 0);
array_splice()
নির্দিষ্ট অবস্থান থেকে উপাদান যোগ/বাদ দেয়।
array_splice($a,1,2);
array_merge()
একাধিক অ্যারে একত্রিত করে।
$c = array_merge($a,$b);
array_combine()
দুটি অ্যারেকে key ও value হিসেবে মিলিয়ে দেয়।
$r = array_combine($k,$v);
Information
Array Information Functions
অ্যারে সম্পর্কে তথ্য জানার ফাংশন।
count()
অ্যারের উপাদান সংখ্যা গণনা করে।
echo count($a); // 3
in_array()
কোনো মান অ্যারেতে আছে কিনা চেক করে।
in_array("Rahim", $a);
array_key_exists()
নির্দিষ্ট key অ্যারেতে আছে কিনা যাচাই করে।
array_key_exists("name",$a);
array_search()
মানের key বা index খুঁজে দেয়।
$k = array_search("Karim",$a);
array_keys()
অ্যারের সব key ফেরত দেয়।
$keys = array_keys($a);
array_values()
অ্যারের সব value ফেরত দেয়।
$vals = array_values($a);
isset()
অ্যারের কোনো key সেট আছে কিনা চেক করে।
isset($a["age"]);
is_array()
ভেরিয়েবলটি অ্যারে কিনা যাচাই করে।
is_array($a); // true/false
Transformation
Array Transformation Functions
অ্যারের ডেটা রূপান্তর বা পরিবর্তন করার ফাংশন।
array_map()
প্রতিটি উপাদানে ফাংশন প্রয়োগ করে নতুন অ্যারে তৈরি করে।
$r = array_map('strtoupper', $a);
array_filter()
শর্ত অনুযায়ী উপাদান ফিল্টার করে।
$r = array_filter($a, fn($x) => $x > 2);
array_reduce()
অ্যারেকে একটি মানে পরিণত করে।
$sum = array_reduce($a, fn($c,$i)=>$c+$i, 0);
array_flip()
key ও value পরস্পর অদলবদল করে।
$r = array_flip($a);
array_reverse()
অ্যারের ক্রম উল্টে দেয়।
$r = array_reverse($a);
array_unique()
ডুপ্লিকেট মান বাদ দিয়ে অনন্য মান রাখে।
$r = array_unique($a);
array_chunk()
অ্যারেকে ছোট ছোট ভাগে বিভক্ত করে।
$r = array_chunk($a, 2);
array_slice()
অ্যারের নির্দিষ্ট অংশ বের করে।
$r = array_slice($a, 1, 3);
Sorting
Array Sorting Functions
অ্যারের উপাদান বিভিন্নভাবে সাজানোর ফাংশন।
sort()
ছোট থেকে বড় ক্রমে সাজায় (index রিসেট হয়)।
sort($a); // [1,2,3,4,5]
rsort()
বড় থেকে ছোট ক্রমে সাজায়।
rsort($a); // [5,4,3,2,1]
asort()
value অনুযায়ী সাজায়, key বজায় রাখে।
asort($a);
arsort()
value অনুযায়ী উল্টো ক্রমে সাজায়, key বজায় রাখে।
arsort($a);
ksort()
key অনুযায়ী ক্রমে সাজায়।
ksort($a);
krsort()
key অনুযায়ী উল্টো ক্রমে সাজায়।
krsort($a);
usort()
কাস্টম ফাংশন দিয়ে সাজায়।
usort($a, fn($x,$y)=>$x<=>$y);
array_multisort()
একাধিক অ্যারে বা কলাম একসাথে সাজায়।
array_multisort($a, SORT_ASC);
Iteration
Array Iteration Functions
অ্যারের প্রতিটি উপাদারের উপর কাজ করার ফাংশন।
foreach
অ্যারের প্রতিটি উপাদান লুপ করার সবচেয়ে সহজ উপায়।
foreach($a as $k=>$v){}
array_walk()
প্রতিটি উপাদারে একটি ফাংশন প্রয়োগ করে (মূল অ্যারে পরিবর্তন করে)।
array_walk($a, fn(&$v)=>$v *= 2);
array_walk_recursive()
Nested অ্যারের প্রতিটি উপাদারে ফাংশন প্রয়োগ করে।
array_walk_recursive($a, $fn);
current() / next() / prev()
অ্যারে পয়েন্টার দিয়ে উপাদার নেভিগেট করে।
current($a); next($a);
reset() / end()
পয়েন্টার শুরুতে বা শেষে নিয়ে যায়।
reset($a); end($a);
list()
অ্যারের মান সরাসরি ভেরিয়েবলে assign করে।
list($a,$b) = [1,2];
Multidimensional
Multidimensional Arrays Functions
বহুমাত্রিক অ্যারে নিয়ে কাজ করার ফাংশন।
array_column()
2D অ্যারে থেকে নির্দিষ্ট কলামের সব মান বের করে।
$names = array_column($students, "name");
array_map() (nested)
Nested অ্যারের প্রতিটি উপাদারে ফাংশন প্রয়োগ করে।
array_map(fn($r)=>array_map('strtoupper',$r),$a);
usort() (nested)
2D অ্যারেকে নির্দিষ্ট কলাম অনুযায়ী সাজায়।
usort($s, fn($a,$b)=>$a["age"]<=>$b["age"]);
array_multisort()
Multidimensional অ্যারে একাধিক column দিয়ে সাজায়।
array_multisort(array_column($s,"age"),SORT_ASC,$s);
foreach (nested)
ভেতরের অ্যারে অ্যাক্সেস করতে nested foreach ব্যবহার হয়।
foreach($m as $row){foreach($row as $v){}}
array_filter() (nested)
2D অ্যারে থেকে শর্ত অনুযায়ী সারি ফিল্টার করে।
array_filter($s, fn($r)=>$r["age"]>20);
Mathematics
Array Mathematics Functions
অ্যারের সংখ্যার উপর গাণিতিক কাজ করার ফাংশন।
array_sum()
অ্যারের সব সংখ্যার যোগফল বের করে।
echo array_sum([1,2,3]); // 6
array_product()
অ্যারের সব সংখ্যার গুণফল বের করে।
echo array_product([1,2,3]); // 6
min()
অ্যারের সবচেয়ে ছোট মান খুঁজে দেয়।
echo min([3,1,4]); // 1
max()
অ্যারের সবচেয়ে বড় মান খুঁজে দেয়।
echo max([3,1,4]); // 4
array_count_values()
প্রতিটি মান কতবার আছে তা গণনা করে।
array_count_values(["a","b","a"]);
range()
একটি নির্দিষ্ট পরিসরের সংখ্যার অ্যারে তৈরি করে।
$r = range(1, 10);
Utility
Array Utility Functions
বিভিন্ন কাজের জন্য সহায়ক ফাংশন।
array_fill()
একটি নির্দিষ্ট মান দিয়ে অ্যারে পূরণ করে।
$r = array_fill(0, 5, "PHP");
array_pad()
অ্যারেকে নির্দিষ্ট দৈর্ঘ্যে বাড়িয়ে দেয়।
$r = array_pad($a, 5, 0);
array_diff()
দুটি অ্যারের পার্থক্য বের করে।
$r = array_diff($a, $b);
array_intersect()
দুটি অ্যারের সাধারণ মান বের করে।
$r = array_intersect($a, $b);
compact()
ভেরিয়েবল নাম ব্যবহার করে associative array তৈরি করে।
$r = compact("name", "age");
extract()
associative array থেকে ভেরিয়েবল তৈরি করে।
extract($student);
shuffle()
অ্যারের উপাদান এলোমেলো করে দেয়।
shuffle($a);
array_rand()
অ্যারে থেকে র‍্যান্ডম key বের করে।
$k = array_rand($a, 2);
